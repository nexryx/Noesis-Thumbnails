cmake_minimum_required(VERSION 3.20)
project(noesis_thumbnails LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0602)
add_library(NoesisThumbnailProvider SHARED src/provider.cpp src/provider.def)
add_executable(noesis-thumbnails src/worker.cpp)
add_executable(native-tests tests/native_tests.cpp)
foreach(target NoesisThumbnailProvider noesis-thumbnails native-tests)
  target_include_directories(${target} PRIVATE src)
  target_link_libraries(${target} PRIVATE ole32 shell32 shlwapi bcrypt gdi32 uuid)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endforeach()
if(MINGW)
  target_link_options(noesis-thumbnails PRIVATE -municode)
  target_link_options(native-tests PRIVATE -municode)
endif()
enable_testing()
add_test(NAME native-tests COMMAND native-tests)
set(NOESIS_SDK "" CACHE PATH "Extracted Noesis pluginsource.zip directory")
if(NOESIS_SDK)
  add_library(noesis_thumbnails_formats_x86_64 SHARED noesis/format_inventory.cpp)
  target_include_directories(noesis_thumbnails_formats_x86_64 PRIVATE "${NOESIS_SDK}")
  target_compile_definitions(noesis_thumbnails_formats_x86_64 PRIVATE _NOE64)
  set_target_properties(noesis_thumbnails_formats_x86_64 PROPERTIES PREFIX "")
endif()
